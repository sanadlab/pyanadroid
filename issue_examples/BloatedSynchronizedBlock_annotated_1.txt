private final List<String> sharedList = new ArrayList<>();
public void addFormattedData(String rawData) {
    // Good: Prepare the data first using only local variables.
    String trimmedData = rawData.trim();
    String upperCaseData = trimmedData.toUpperCase();

    // The synchronized block is now minimal, reducing contention.
    synchronized (sharedList) {
        sharedList.add(upperCaseData);
    }
}
