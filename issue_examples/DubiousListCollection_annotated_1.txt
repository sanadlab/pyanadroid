// Bad: If this list has thousands of admins, .contains() will be very slow (O(n)).
private final List<String> adminUsernames = Arrays.asList("root", "admin", "superuser", ...);

public boolean isAdmin(String username) {
    return adminUsernames.contains(username);
}

-----

// Good: Using a Set provides fast, average O(1) lookups.
private final Set<String> adminUsernames = new HashSet<>(Arrays.asList("root", "admin", "superuser", ...));

public boolean isAdmin(String username) {
    return adminUsernames.contains(username);
}
