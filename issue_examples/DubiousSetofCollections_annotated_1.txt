// Bad: The hashCode of a Set depends on its contents. If an inner set is modified
// after being added to the outer set, it can become "lost" and break the contract.
Set<Set<String>> tagSets = new HashSet<>();
Set<String> post1Tags = new HashSet<>(Arrays.asList("java", "performance"));
tagSets.add(post1Tags);

// This modification changes the hashCode of post1Tags, breaking the outer set's invariants.
post1Tags.add("optimization"); 

-----

// Good: If you must store collections, ensure they are immutable before adding them.
Set<Set<String>> tagSets = new HashSet<>();
Set<String> post1Tags = new HashSet<>(Arrays.asList("java", "performance"));
tagSets.add(Collections.unmodifiableSet(post1Tags));
