// Inefficient: Exponential complexity due to repeated calculations
fun fibonacci(n: Int): Int {
    if (n <= 1) {
        return n
    }
    return fibonacci(n - 1) + fibonacci(n - 2)
}

// Efficient: Iterative approach with linear complexity O(n)
fun fibonacci(n: Int): Int {
    if (n <= 1) return n
    var a = 0
    var b = 1
    for (i in 2..n) {
        val sum = a + b
        a = b
        b = sum
    }
    return b
}
