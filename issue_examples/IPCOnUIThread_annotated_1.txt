// In an Android Activity or Fragment
fun onFindContactsClicked(view: View) {
    // Good: The blocking call is moved to a background thread (IO dispatcher)
    lifecycleScope.launch {
        val cursor = withContext(Dispatchers.IO) {
            contentResolver.query(
                ContactsContract.Contacts.CONTENT_URI,
                null, null, null, null
            )
        }
        // Update UI with cursor data on the main thread...
    }
}
