// Inefficient: A second lookup is performed inside the loop for each key
for (productId in productPrices.keys) {
    val currentPrice = productPrices[productId]
    if (currentPrice != null && currentPrice < 20.00) {
        // apply a 10% discount
        productPrices[productId] = currentPrice * 0.9
    }
}

---- 

// Efficient: Uses entrySet to get direct access to the value and update it
for (entry in productPrices.entries) {
    if (entry.value < 20.00) {
        // apply a 10% discount
        entry.setValue(entry.value * 0.9)
    }
}
