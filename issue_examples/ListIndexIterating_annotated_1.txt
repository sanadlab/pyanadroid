// Bad: For a LinkedList, list.get(i) is an O(n) operation. This makes the
// entire loop O(n^2), which is extremely slow for large lists.
List<String> names = new LinkedList<>();
// ... add many names ...
for (int i = 0; i < names.size(); i++) {
    System.out.println(names.get(i));
}

-----

List<String> names = new LinkedList<>();
// ... add many names ...
// Good: The for-each loop uses an iterator internally, which is O(n) for a LinkedList.
for (String name : names) {
    System.out.println(name);
}
