// Good: The Pattern is compiled once into a static final field, the most efficient approach.
private static final Pattern EMAIL_PATTERN =
    Pattern.compile("^[\\w.-]+@[\\w.-]+\\.[a-zA-Z]{2,7}$");

public int countValidEmails(List<String> emails) {
    int validCount = 0;
    for (String email : emails) {
        // The pre-compiled pattern is reused.
        if (EMAIL_PATTERN.matcher(email).matches()) {
            validCount++;
        }
    }
    return validCount;
}
