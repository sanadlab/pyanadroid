fun onValidateInputClicked(view: View) {
    val userInput = editText.text.toString()
    val complexRegex = "^(?=.*[0-9])(?=.*[a-z])(?=.*[A-Z])(?=.*[@#$%^&+=])(?=\\S+$).{8,}$"

    // Good: The expensive regex matching is offloaded from the UI thread
    lifecycleScope.launch {
        val isValid = withContext(Dispatchers.Default) { // Use Default for CPU-bound work
            userInput.matches(complexRegex.toRegex())
        }
        if (!isValid) {
            showError("Input is not valid.")
        }
    }
}
