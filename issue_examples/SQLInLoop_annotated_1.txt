// Bad: If userIds has 1000 IDs, this executes 1000 separate queries to the database.
// This is known as the "N+1 query problem".
public List<User> getUsers(List<Integer> userIds) {
    List<User> users = new ArrayList<>();
    for (Integer id : userIds) {
        // A new query is sent for each ID.
        users.add(queryUserById(id));
    }
    return users;
}
