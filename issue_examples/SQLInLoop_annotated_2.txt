// Bad: High network latency and database overhead from running many small delete statements.
fun deleteProducts(conn: Connection, productIds: List<String>) {
    for (id in productIds) {
        conn.prepareStatement("DELETE FROM products WHERE id = ?").use {
            it.setString(1, id)
            it.executeUpdate()
        }
    }
}

----

// Good: Use JDBC batching to send all delete commands in a single round-trip.
fun deleteProducts(conn: Connection, productIds: List<String>) {
    conn.prepareStatement("DELETE FROM products WHERE id = ?").use { stmt ->
        for (id in productIds) {
            stmt.setString(1, id)
            stmt.addBatch()
        }
        stmt.executeBatch()
    }
}
