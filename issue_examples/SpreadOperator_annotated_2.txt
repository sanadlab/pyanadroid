val firstHalf = intArrayOf(1, 2, 3)
val secondHalf = intArrayOf(4, 5, 6)
// Bad: Creates a copy of firstHalf and a copy of secondHalf to create the new array.
val combined = intArrayOf(*firstHalf, *secondHalf)

val firstHalf = intArrayOf(1, 2, 3)
val secondHalf = intArrayOf(4, 5, 6)
// Good: The plus operator is often optimized for this use case and is clearer.
val combined = firstHalf + secondHalf
