val defaultScore: Float = 0.0f

val nullableScore: Int? = null
val defaultScore: Float = 0.0f

// Good: Be explicit about the desired type to avoid unexpected coercion.
val finalScore: Number = nullableScore?.toFloat() ?: defaultScore
