class ReportGenerator {
    private var totalRevenue: Double = 0.0

    fun addSale(amount: Double) {
        this.totalRevenue += amount
    }

    // Good: A getter is added so the value can be read and used.
    fun getTotalRevenue(): Double {
        return this.totalRevenue
    }

    fun generateReport() {
        // Now it can be used:
        val revenue = getTotalRevenue()
        // ...
    }
}
